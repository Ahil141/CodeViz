### DESCRIPTION: Python implementation of a circular linked list with insert, delete, and traversal operations.

class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

class CircularLinkedList:
    def __init__(self):
        self.head = None

    def insert_at_end(self, data):
        new_node = Node(data)
        if not self.head:
            self.head = new_node
            new_node.next = new_node
            return
        curr = self.head
        while curr.next != self.head:
            curr = curr.next
        curr.next = new_node
        new_node.next = self.head

    def insert_at_start(self, data):
        self.insert_at_end(data)
        self.head = self.head.next

    def delete_at_start(self):
        if not self.head:
            return
        curr = self.head
        while curr.next != self.head:
            curr = curr.next
        if self.head == curr:
            self.head = None
        else:
            curr.next = self.head.next
            self.head = self.head.next

    def traverse(self):
        if not self.head:
            print("Empty List")
            return
        curr = self.head
        while True:
            print(curr.data, end=" -> ")
            curr = curr.next
            if curr == self.head:
                break
        print("(back to head)")
