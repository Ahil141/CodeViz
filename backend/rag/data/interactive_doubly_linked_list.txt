class Node { 
    constructor(data, next = null, prev = null) { 
        this.data = data; 
        this.next = next; 
        this.prev = prev; 
    } 
}

class DoublyLinkedList {
    constructor() { this.head = null; this.tail = null; this.size = 0; }

    insertFirst(data) {
        const node = new Node(data, this.head, null);
        if (this.head) this.head.prev = node;
        this.head = node;
        if (!this.tail) this.tail = node;
        this.size++;
    }

    insertLast(data) {
        const node = new Node(data, null, this.tail);
        if (!this.tail) { this.head = this.tail = node; }
        else { this.tail.next = node; this.tail = node; }
        this.size++;
    }

    insertAt(data, i) {
        if (i < 0 || i > this.size) return;
        if (i === 0) return this.insertFirst(data);
        if (i === this.size) return this.insertLast(data);

        let c = this.head, count = 0;
        while (count < i) { c = c.next; count++; }
        const node = new Node(data, c, c.prev);
        c.prev.next = node;
        c.prev = node;
        this.size++;
    }

    removeFirst() {
        if (!this.head) return;
        this.head = this.head.next;
        if (this.head) this.head.prev = null;
        else this.tail = null;
        this.size--;
    }

    removeLast() {
        if (!this.tail) return;
        this.tail = this.tail.prev;
        if (this.tail) this.tail.next = null;
        else this.head = null;
        this.size--;
    }

    removeAt(i) {
        if (i < 0 || i >= this.size) return;
        if (i === 0) return this.removeFirst();
        if (i === this.size - 1) return this.removeLast();

        let c = this.head, count = 0;
        while (count < i) { c = c.next; count++; }
        c.prev.next = c.next;
        c.next.prev = c.prev;
        this.size--;
    }
}

const list = new DoublyLinkedList();
const visualizer = document.getElementById('visualizer');
const headDisplay = document.getElementById('head-display');
const valueInput = document.getElementById('nodeValue');
const indexInput = document.getElementById('nodeIndex');

function getValue() { 
    const v = valueInput.value.trim(); 
    if (v === '') { alert('Please enter a value.'); return null; } 
    valueInput.value = ''; 
    return isNaN(v) ? v : Number(v); 
}

function getIndex() { 
    const i = parseInt(indexInput.value.trim(), 10); 
    if (isNaN(i) || i < 0) { alert('Please enter a valid index.'); return null; } 
    indexInput.value = ''; 
    return i; 
}

function updateUI() {
    visualizer.innerHTML = '';
    let current = list.head;

    if (!current) {
        headDisplay.textContent = 'HEAD → NULL';
        visualizer.innerHTML = '<span class="null-text">List is Empty</span>';
        return;
    }

    headDisplay.textContent = `HEAD → Node(${list.head.data})`;
    let first = true;

    while (current) {
        const nodeDiv = document.createElement('div');
        nodeDiv.className = 'node';
        nodeDiv.textContent = current.data;

        if (first) {
            const headLabel = document.createElement('div');
            headLabel.className = 'head-label';
            headLabel.textContent = 'HEAD';
            nodeDiv.appendChild(headLabel);
            first = false;
        }

        visualizer.appendChild(nodeDiv);

        if (current.next) {
            const arrowSpan = document.createElement('span');
            arrowSpan.className = 'arrow';
            arrowSpan.innerHTML = '&larr;&rarr;';
            visualizer.appendChild(arrowSpan);
        }

        current = current.next;
    }

    const nullSpan = document.createElement('span');
    nullSpan.className = 'null-text';
    nullSpan.innerHTML = '&rarr; NULL';
    visualizer.appendChild(nullSpan);
}
