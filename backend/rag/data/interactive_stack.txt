<script>
class Stack {
    constructor() {
        this.items = [];
    }
    push(x) { this.items.push(x); }
    pop() { return this.items.pop(); }
    peek() { return this.items[this.items.length - 1]; }
    clear() { this.items = []; }
    isEmpty() { return this.items.length === 0; }
}

const stack = new Stack();
const visualizer = document.getElementById("visualizer");
const topDisplay = document.getElementById("stackTop");
const valueInput = document.getElementById("valueInput");

function getValue() {
    const v = valueInput.value.trim();
    if (v === "") { alert("Please enter a value."); return null; }
    valueInput.value = "";
    return isNaN(v) ? v : Number(v);
}

function updateUI() {
    visualizer.innerHTML = "";

    if (stack.isEmpty()) {
        topDisplay.textContent = "TOP → NULL";
        visualizer.innerHTML = '<span style="color:#7f8c8d;font-size:20px;font-style:italic;">Stack is Empty</span>';
        return;
    }

    topDisplay.textContent = `TOP → ${stack.peek()}`;

    stack.items.forEach((val, index) => {
        const div = document.createElement("div");
        div.className = "stack-item";
        div.textContent = val;

        if (index === stack.items.length - 1) {
            const label = document.createElement("div");
            label.className = "top-label";
            label.textContent = "TOP";
            div.appendChild(label);
        }

        visualizer.appendChild(div);
    });
}

document.getElementById("pushBtn").addEventListener("click", () => {
    const v = getValue();
    if (v !== null) { stack.push(v); updateUI(); }
});

document.getElementById("popBtn").addEventListener("click", () => {
    if (stack.isEmpty()) { alert("Stack is empty!"); return; }
    stack.pop();
    updateUI();
});

document.getElementById("peekBtn").addEventListener("click", () => {
    if (stack.isEmpty()) { alert("Stack is empty!"); return; }
    alert("Top element = " + stack.peek());
});

document.getElementById("clearBtn").addEventListener("click", () => {
    stack.clear();
    updateUI();
});

/* Modal for Python Code */
document.getElementById("showCodeBtn").addEventListener("click", () => {
    const code = `
class Stack:
    def __init__(self):
        self.items = []

    def push(self, x):
        self.items.append(x)

    def pop(self):
        if not self.items:
            return None
        return self.items.pop()

    def peek(self):
        if not self.items:
            return None
        return self.items[-1]

    def clear(self):
        self.items = []

    def is_empty(self):
        return len(self.items) == 0
    `;
    document.getElementById("pythonCode").textContent = code.trim();
    document.getElementById("codeModal").style.display = "flex";
});

updateUI();
</script>
