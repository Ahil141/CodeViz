### TOPIC: QUEUE
### DESCRIPTION: Simple Python implementation of a queue using a list with enqueue, dequeue, and is_empty operations.

class Queue:
    def __init__(self):
        self.items = []

    def enqueue(self, item):
        self.items.append(item)

    def dequeue(self):
        if not self.is_empty():
            return self.items.pop(0)
        return None

    def is_empty(self):
        return len(self.items) == 0
